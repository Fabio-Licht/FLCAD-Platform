cmake_minimum_required(VERSION 3.24)
project(cad_asset_fs LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "cad_asset_fs v1 requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(cad_asset_fs SHARED src/cad_asset_fs.cpp)
target_include_directories(cad_asset_fs PUBLIC include)
target_compile_definitions(cad_asset_fs PRIVATE CAF_BUILD UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(cad_asset_fs PRIVATE bcrypt)
if(MSVC)
  target_compile_options(cad_asset_fs PRIVATE /W4 /WX /permissive- /EHsc /analyze /utf-8)
endif()
include(CTest)
if(BUILD_TESTING AND PROJECT_IS_TOP_LEVEL)
  add_library(cad_asset_fs_source_test SHARED tests/source_damage.cpp)
  target_include_directories(cad_asset_fs_source_test PRIVATE include)
  target_compile_definitions(cad_asset_fs_source_test PRIVATE CAF_BUILD UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(cad_asset_fs_source_test PRIVATE bcrypt)
  add_library(cad_asset_fs_incompatible SHARED tests/abi_incompatible.c)
  add_executable(cad_asset_fs_tests tests/test_main.cpp tests/abi_c.c)
  target_link_libraries(cad_asset_fs_tests PRIVATE cad_asset_fs advapi32 bcrypt)
  target_compile_definitions(cad_asset_fs_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(cad_asset_fs_tests PRIVATE /W4 /WX /EHsc /utf-8)
  endif()
  foreach(case gateway basic names static_junction symlink races collision rename sharing denied invalid leaks unicode concurrency termination volume_policy volume)
    add_test(NAME caf_${case} COMMAND cad_asset_fs_tests ${case})
    set_tests_properties(caf_${case} PROPERTIES TIMEOUT 60)
  endforeach()
  set_tests_properties(caf_volume PROPERTIES SKIP_RETURN_CODE 77)
  set_tests_properties(caf_symlink PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(cad_asset_fs_fault_tests tests/test_main.cpp tests/abi_c.c src/cad_asset_fs.cpp)
  target_include_directories(cad_asset_fs_fault_tests PRIVATE include)
  target_compile_definitions(cad_asset_fs_fault_tests PRIVATE CAF_BUILD CAF_TEST_INSTRUMENTATION UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(cad_asset_fs_fault_tests PRIVATE bcrypt advapi32)
  if(MSVC)
    target_compile_options(cad_asset_fs_fault_tests PRIVATE /W4 /WX /EHsc /utf-8)
  endif()
  foreach(case inplace_create inplace_rename faults)
    add_test(NAME caf_${case} COMMAND cad_asset_fs_fault_tests ${case})
    set_tests_properties(caf_${case} PROPERTIES TIMEOUT 60)
  endforeach()
endif()
