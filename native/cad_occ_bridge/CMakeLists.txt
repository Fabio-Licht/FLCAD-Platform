cmake_minimum_required(VERSION 3.24)
project(cad_occ_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(cad_occ_bridge SHARED src/cad_occ_bridge.cpp)
target_include_directories(cad_occ_bridge PUBLIC include ../cad_asset_fs/include ../opencascade/include)
target_compile_definitions(cad_occ_bridge PRIVATE COB_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(cad_occ_bridge PRIVATE /W4 /WX /analyze /EHsc /utf-8)
endif()

include(CTest)
if(BUILD_TESTING AND PROJECT_IS_TOP_LEVEL)
  add_library(cad_occ_bridge_incompatible SHARED tests/incompatible.cpp)
  add_library(cad_occ_bridge_test SHARED src/cad_occ_bridge.cpp tests/control.cpp)
  target_include_directories(cad_occ_bridge_test PRIVATE include ../cad_asset_fs/include ../opencascade/include)
  target_compile_definitions(cad_occ_bridge_test PRIVATE COB_BUILD COB_TESTING WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(cad_occ_bridge_smoke tests/smoke.cpp)
  target_link_libraries(cad_occ_bridge_smoke PRIVATE cad_occ_bridge)
  add_executable(cad_occ_bridge_fault_tests tests/smoke.cpp src/cad_occ_bridge.cpp)
  target_include_directories(cad_occ_bridge_fault_tests PRIVATE include ../cad_asset_fs/include ../opencascade/include)
  target_compile_definitions(cad_occ_bridge_fault_tests PRIVATE COB_BUILD COB_TESTING WIN32_LEAN_AND_MEAN NOMINMAX)
  set(COB_CAF_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build/cad_asset_fs" CACHE PATH "CAF build")
  set(COB_OCC_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build/occ_mesh_stream" CACHE PATH "OCCT build")
  foreach(t cad_occ_bridge_smoke cad_occ_bridge_fault_tests)
    target_compile_options(${t} PRIVATE /utf-8)
    add_test(NAME ${t} COMMAND ${t} "${COB_CAF_DIR}/$<CONFIG>/cad_asset_fs.dll" "${COB_OCC_DIR}/$<CONFIG>/flcad_opencascade.dll")
    set_tests_properties(${t} PROPERTIES TIMEOUT 120)
  endforeach()
endif()
